(defun my/ts-node-find-child-of-type (node type)
  (let* ((n (treesit-node-child-count node))
         (i 0) (hit nil))
    (while (and (< i n) (not hit))
      (let ((c (treesit-node-child node i t)))
        (when (and c (string= (treesit-node-type c) type))
          (setq hit c)))
      (setq i (1+ i)))
    hit))

(defun my/ts-node-any-descendant-p (node pred)
  (let ((n (treesit-node-child-count node))
        (ok (funcall pred node)))
    (let ((i 0))
      (while (and (not ok) (< i n))
        (setq ok (my/ts-node-any-descendant-p (treesit-node-child node i t) pred))
        (setq i (1+ i))))
    ok))

(defun my/ts-node-has-ancestor-type (node type)
  (let ((p (treesit-node-parent node)) (found nil))
    (while (and p (not found))
      (setq found (string= (treesit-node-type p) type))
      (setq p (treesit-node-parent p)))
    found))

(defun my/ts-trim-braces-range (block-node)
  (let ((beg (1+ (treesit-node-start block-node)))
        (end (1- (treesit-node-end   block-node))))
    (when (< beg end) (cons beg end))))

(defun my/ts-fold-range-rust-method-body (node _offset)
  (let* ((body (treesit-node-child-by-field-name node "body"))
         (params (treesit-node-child-by-field-name node "parameters"))
         (has-self (and params
                        (my/ts-node-any-descendant-p
                         params (lambda (n) (string= (treesit-node-type n) "self_parameter")))))
         (in-impl (my/ts-node-has-ancestor-type node "impl_item")))
    (when (and body (or has-self in-impl))
      (my/ts-trim-braces-range body))))

(defun my/ts-fold-range-zig-fn-body (node _offset)
  (let ((has-proto (my/ts-node-find-child-of-type node "FnProto"))
        (block     (my/ts-node-find-child-of-type node "Block")))
    (when (and has-proto block)
      (my/ts-trim-braces-range block))))

** Use =difft= for diffing
Taken from [[https://www.tsdh.org/posts/2022-08-01-difftastic-diffing-with-magit.html][this article.]]

#+begin_src emacs-lisp
(after! magit
 (defun dysthesis/magit--with-difftastic (buffer command)
   "Run COMMAND with GIT_EXTERNAL_DIFF=difft then show result in BUFFER."
   (let ((process-environment
          (cons (concat "GIT_EXTERNAL_DIFF=difft --width="
                        (number-to-string (frame-width)))
                process-environment)))
     ;; Clear the result buffer (we might regenerate a diff, e.g., for
     ;; the current changes in our working directory).
     (with-current-buffer buffer
       (setq buffer-read-only nil)
       (erase-buffer))
     ;; Now spawn a process calling the git COMMAND.
     (make-process
      :name (buffer-name buffer)
      :buffer buffer
      :command command
      ;; Don't query for running processes when emacs is quit.
      :noquery t
      ;; Show the result buffer once the process has finished.
      :sentinel (lambda (proc event)
                  (when (eq (process-status proc) 'exit)
                    (with-current-buffer (process-buffer proc)
                      (goto-char (point-min))
                      (ansi-color-apply-on-region (point-min) (point-max))
                      (setq buffer-read-only t)
                      (view-mode)
                      (end-of-line)
                      ;; difftastic diffs are usually 2-column side-by-side,
                      ;; so ensure our window is wide enough.
                      (let ((width (current-column)))
                        (while (zerop (forward-line 1))
                          (end-of-line)
                          (setq width (max (current-column) width)))
                        ;; Add column size of fringes
                        (setq width (+ width
                                       (fringe-columns 'left)
                                       (fringe-columns 'right)))
                        (goto-char (point-min))
                        (pop-to-buffer
                         (current-buffer)
                         `(;; If the buffer is that wide that splitting the frame in
                           ;; two side-by-side windows would result in less than
                           ;; 80 columns left, ensure it's shown at the bottom.
                           ,(when (> 80 (- (frame-width) width))
                              #'display-buffer-at-bottom)
                           (window-width
                            . ,(min width (frame-width)))))))))))

  (defun dysthesis/magit-show-with-difftastic (rev)
   "Show the result of \"git show REV\" with GIT_EXTERNAL_DIFF=difft."
   (interactive
    (list (or
           ;; If REV is given, just use it.
           (when (boundp 'rev) rev)
           ;; If not invoked with prefix arg, try to guess the REV from
           ;; point's position.
           (and (not current-prefix-arg)
                (or (magit-thing-at-point 'git-revision t)
                    (magit-branch-or-commit-at-point)))
           ;; Otherwise, query the user.
           (magit-read-branch-or-commit "Revision"))))
   (if (not rev)
       (error "No revision specified")
     (dysthesis/magit--with-difftastic
      (get-buffer-create (concat "*git show difftastic " rev "*"))
      (list "git" "--no-pager" "show" "--ext-diff" rev)))))
 (defun dysthesis/magit-diff-with-difftastic (arg)
   "Show the result of \"git diff ARG\" with GIT_EXTERNAL_DIFF=difft."
   (interactive
    (list (or
           ;; If RANGE is given, just use it.
           (when (boundp 'range) range)
           ;; If prefix arg is given, query the user.
           (and current-prefix-arg
                (magit-diff-read-range-or-commit "Range"))
           ;; Otherwise, auto-guess based on position of point, e.g., based on
           ;; if we are in the Staged or Unstaged section.
           (pcase (magit-diff--dwim)
             ('unmerged (error "unmerged is not yet implemented"))
             ('unstaged nil)
             ('staged "--cached")
             (`(stash . ,value) (error "stash is not yet implemented"))
             (`(commit . ,value) (format "%s^..%s" value value))
             ((and range (pred stringp)) range)
             (_ (magit-diff-read-range-or-commit "Range/Commit"))))))
   (let ((name (concat "*git diff difftastic"
                       (if arg (concat " " arg) "")
                       "*")))
     (dysthesis/magit--with-difftastic
      (get-buffer-create name)
      `("git" "--no-pager" "diff" "--ext-diff" ,@(when arg (list arg))))))


 (transient-define-prefix dysthesis/magit-aux-commands ()
   "My personal auxiliary magit commands."
   ["Auxiliary commands"
    ("d" "Difftastic Diff (dwim)" dysthesis/magit-diff-with-difftastic)
    ("s" "Difftastic Show" dysthesis/magit-show-with-difftastic)])

 (transient-append-suffix 'magit-dispatch "!"
   '("#" "My Magit Cmds" dysthesis/magit-aux-commands))

 (define-key magit-status-mode-map (kbd "#") #'dysthesis/magit-aux-commands))

  (defun my/org-latex-preview-center (ov)

        (overlay-put ov 'justify justify)
        (overlay-put ov 'before-string (overlay-get ov 'justify)))))
